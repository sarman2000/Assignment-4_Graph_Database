// One-panel summary
CALL { MATCH (n) RETURN count(n) AS totalNodes }
CALL { MATCH ()-[r]->() RETURN count(r) AS totalRelationships }
CALL { MATCH (n) UNWIND labels(n) AS l RETURN count(DISTINCT l) AS labelsUsed }
CALL { MATCH ()-[r]->() RETURN count(DISTINCT type(r)) AS relationshipTypes }
RETURN totalNodes, totalRelationships, labelsUsed, relationshipTypes;

// ===== Part B: integrity and business-rule checks (all violations must be 0) =====

MATCH (x:PurchaseOrderLine) WHERE NOT (x)-[:LINE_OF]->(:PurchaseOrder)
RETURN 'PO line without order' AS check, count(x) AS violations
UNION ALL
MATCH (x:PurchaseOrderLine) WHERE NOT (x)-[:FOR_PRODUCT]->(:Product)
RETURN 'PO line without product' AS check, count(x) AS violations
UNION ALL
MATCH (x:PurchaseOrder) WHERE NOT (x)-[:ORDERED_FROM]->(:Supplier)
RETURN 'Purchase order without supplier' AS check, count(x) AS violations
UNION ALL
MATCH (x:PurchaseOrder) WHERE NOT (x)-[:CREATED_BY]->(:ProcurementEmployee)
RETURN 'Purchase order not created by procurement employee' AS check, count(x) AS violations
UNION ALL
MATCH (x:GoodsReceipt) WHERE NOT (x)-[:RECEIPT_FOR]->(:PurchaseOrder)
RETURN 'Goods receipt without order' AS check, count(x) AS violations
UNION ALL
MATCH (x:GoodsReceipt) WHERE NOT (x)-[:RECEIVED_BY]->(:WarehouseEmployee)
RETURN 'Goods receipt not received by warehouse employee' AS check, count(x) AS violations
UNION ALL
MATCH (x:GoodsReceiptLine) WHERE NOT (x)-[:RECEIPT_LINE_OF]->(:GoodsReceipt)
RETURN 'Receipt line without receipt' AS check, count(x) AS violations
UNION ALL
MATCH (x:GoodsReceiptLine) WHERE NOT (x)-[:MATCHES_LINE]->(:PurchaseOrderLine)
RETURN 'Receipt line without PO line' AS check, count(x) AS violations
UNION ALL
MATCH (x:ProductBatch) WHERE NOT (x)-[:RECEIVED_AS]->(:GoodsReceiptLine)
RETURN 'Batch without receipt line' AS check, count(x) AS violations
UNION ALL
MATCH (x:ProductBatch) WHERE NOT (x)-[:OF_PRODUCT]->(:Product)
RETURN 'Batch without OF_PRODUCT shortcut' AS check, count(x) AS violations
UNION ALL
MATCH (x:StorageLocation) WHERE NOT (x)-[:IN_WAREHOUSE]->(:Warehouse)
RETURN 'Storage location without warehouse' AS check, count(x) AS violations
UNION ALL
MATCH (x:Inventory) WHERE NOT (x)-[:OF_BATCH]->(:ProductBatch) OR NOT (x)-[:AT_LOCATION]->(:StorageLocation)
RETURN 'Inventory without batch or location' AS check, count(x) AS violations
UNION ALL
MATCH (x:StockTransaction) WHERE NOT (x)-[:MOVES_BATCH]->(:ProductBatch) OR NOT (x)-[:PERFORMED_BY]->(:WarehouseEmployee)
RETURN 'Transaction without batch or employee' AS check, count(x) AS violations
UNION ALL
MATCH (x:StockTransaction) WHERE NOT (x)-[:FROM_LOCATION]->() AND NOT (x)-[:TO_LOCATION]->()
RETURN 'Transaction with neither FROM nor TO location' AS check, count(x) AS violations
UNION ALL
MATCH (x:StockAdjustment) WHERE NOT (x)-[:ADJUSTS]->(:Inventory) OR NOT (x)-[:PERFORMED_BY]->(:WarehouseEmployee)
RETURN 'Adjustment without inventory or employee' AS check, count(x) AS violations
UNION ALL
MATCH (x:DisposalRecord) WHERE NOT (x)-[:DISPOSES_FROM]->(:Inventory) OR NOT (x)-[:PERFORMED_BY]->(:WarehouseEmployee)
RETURN 'Disposal without inventory or employee' AS check, count(x) AS violations
UNION ALL
MATCH (x:Employee) WHERE NOT (x:ProcurementEmployee XOR x:WarehouseEmployee)
RETURN 'Employee without exactly one subtype label (complete + disjoint)' AS check, count(x) AS violations
UNION ALL
MATCH (x:GoodsReceiptLine)-[:RECEIPT_LINE_OF]->(:GoodsReceipt)-[:RECEIPT_FOR]->(po1:PurchaseOrder) MATCH (x)-[:MATCHES_LINE]->(:PurchaseOrderLine)-[:LINE_OF]->(po2:PurchaseOrder) WHERE po1 <> po2
RETURN 'Same-order rule broken (receipt line matches a line of a different order)' AS check, count(x) AS violations
UNION ALL
MATCH (x:ProductBatch)-[:OF_PRODUCT]->(p1:Product) MATCH (x)-[:RECEIVED_AS]->(:GoodsReceiptLine)-[:MATCHES_LINE]->(:PurchaseOrderLine)-[:FOR_PRODUCT]->(p2:Product) WHERE p1 <> p2
RETURN 'OF_PRODUCT shortcut disagrees with 3-hop path' AS check, count(x) AS violations
UNION ALL
MATCH (x:Inventory) WHERE x.quantityOnHand < 0
RETURN 'Negative quantity on hand' AS check, count(x) AS violations
UNION ALL
MATCH (x:PurchaseOrderLine)<-[:MATCHES_LINE]-(g:GoodsReceiptLine) WITH x, sum(g.quantityReceived) AS received WHERE received > x.quantityOrdered
RETURN 'Received quantity above ordered quantity for a PO line' AS check, count(x) AS violations
UNION ALL
MATCH (x:ProductBatch) WHERE x.expiryDate < x.manufactureDate
RETURN 'Expiry date before manufacture date' AS check, count(x) AS violations;
